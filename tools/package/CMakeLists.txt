cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(amw_delivery VERSION 0.2.0 LANGUAGES CXX)
endif()
if(TARGET amw_project)
  message(FATAL_ERROR "Only one src-gen contract may be added to a build. Merge contracts in the editor.")
endif()
option(AMW_BUILD_CM "Build communication and generated App interfaces" ON)
find_package(Threads REQUIRED)
if(AMW_BUILD_CM)
  include(cmake/Dependencies.cmake)
endif()
if(NOT TARGET amw_time)
  set(AMW_RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/runtime")
  include(cmake/AmwRuntime.cmake)
endif()
if(AMW_BUILD_CM)
  file(GLOB WIRE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/generated/wire/*.cxx")
  add_library(amw_project generated/apps.cpp ${WIRE_SOURCES})
  add_library(amw::app ALIAS amw_project)
  target_include_directories(amw_project PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/generated" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/generated/wire")
  target_compile_features(amw_project PUBLIC cxx_std_17)
  target_link_libraries(amw_project PUBLIC amw_cm amw_log amw_time PRIVATE amw_fastdds)
  set_target_properties(amw_project PROPERTIES POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
  if(UNIX)
    set_target_properties(amw_project PROPERTIES BUILD_RPATH_USE_ORIGIN ON INSTALL_RPATH "$ORIGIN")
  endif()
  if(MSVC)
    target_compile_options(amw_project PRIVATE /utf-8)
  endif()
endif()
include(cmake/Deploy.cmake)
