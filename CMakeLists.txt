cmake_minimum_required(VERSION 3.22)
project(adaptive_middleware VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
include(CTest)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
option(AMW_BUILD_CM "Build CM and Fast DDS binding (OFF builds standalone log/time)" ON)
option(AMW_BUILD_LOG_EXAMPLE "Build the standalone logging example" OFF)

set(AMW_GENERATED_DIR "" CACHE PATH "Complete src-gen delivery directory")
if(AMW_BUILD_CM AND AMW_GENERATED_DIR)
  include("${AMW_GENERATED_DIR}/cmake/Dependencies.cmake")
endif()
set(AMW_RUNTIME_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
include(cmake/AmwRuntime.cmake)
if(AMW_BUILD_CM)
  if(AMW_GENERATED_DIR)
    add_subdirectory("${AMW_GENERATED_DIR}" generated)
  endif()
  if(BUILD_TESTING)
    add_executable(cm_runtime_test tests/runtime_test.cpp)
    target_link_libraries(cm_runtime_test PRIVATE amw_cm amw_log)
    add_test(NAME cm_runtime_logging COMMAND cm_runtime_test "${CMAKE_CURRENT_SOURCE_DIR}/docs/app-guide/examples/logging.xml")
    set_tests_properties(cm_runtime_logging PROPERTIES TIMEOUT 20)
    add_test(NAME cm_runtime COMMAND cm_runtime_test)
    set_tests_properties(cm_runtime PROPERTIES TIMEOUT 20)
    add_executable(cm_matches_test tests/matches_test.cpp)
    target_link_libraries(cm_matches_test PRIVATE amw_cm)
    add_test(NAME cm_matches COMMAND cm_matches_test)
    set_tests_properties(cm_matches PROPERTIES TIMEOUT 15)
  endif()
  install(TARGETS amw_cm amw_fastdds EXPORT amwCmTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(EXPORT amwCmTargets NAMESPACE amw:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/amw)
endif()
if(BUILD_TESTING)
  add_executable(log_runtime_test tests/log_test.cpp)
  target_link_libraries(log_runtime_test PRIVATE amw_log Threads::Threads)
  add_test(NAME log_runtime COMMAND log_runtime_test "${CMAKE_CURRENT_BINARY_DIR}/log-test.xml")
  set_tests_properties(log_runtime PROPERTIES TIMEOUT 30)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME log_backpressure COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/log_backpressure.py" $<TARGET_FILE:log_runtime_test>)
    set_tests_properties(log_backpressure PROPERTIES TIMEOUT 40)
  endif()
  foreach(target cm_runtime_test cm_matches_test log_runtime_test)
    if(TARGET ${target})
      if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8)
      else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
      endif()
    endif()
  endforeach()
endif()
if(AMW_BUILD_LOG_EXAMPLE)
  add_executable(amw_log_example docs/app-guide/examples/logging.cpp)
  target_link_libraries(amw_log_example PRIVATE amw_log)
endif()
install(TARGETS amw_time amw_log EXPORT amwTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT amwTargets NAMESPACE amw:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/amw)
install(FILES third_party/tinyxml2/LICENSE.txt third_party/tinyxml2/README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/amw/licenses/tinyxml2)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/amwConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/amwConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/amw)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/amwConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/amwConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/amwConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/amw)
option(AMW_BUILD_DEMO "Build the bundled demo for its generated schema" OFF)
if(AMW_BUILD_DEMO AND TARGET amw_project)
  add_executable(amw_producer examples/demo/producer.cpp)
  add_executable(amw_consumer examples/demo/consumer.cpp)
  target_link_libraries(amw_producer PRIVATE amw_project)
  target_link_libraries(amw_consumer PRIVATE amw_project)
endif()
