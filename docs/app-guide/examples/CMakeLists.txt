cmake_minimum_required(VERSION 3.22)
project(amw_guide_examples LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(AMW_GENERATED_DIR "" CACHE PATH "完整的 demo src-gen 交付目录")
if(NOT EXISTS "${AMW_GENERATED_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "请指定 AMW_GENERATED_DIR 为完整的 demo 生成目录")
endif()
add_subdirectory("${AMW_GENERATED_DIR}" generated)
add_executable(guide_publisher publisher.cpp)
add_executable(guide_subscriber subscriber.cpp)
target_link_libraries(guide_publisher PRIVATE amw::app)
target_link_libraries(guide_subscriber PRIVATE amw::app)

add_executable(guide_logging logging.cpp)
target_link_libraries(guide_logging PRIVATE amw::log amw::time)
foreach(app guide_publisher guide_subscriber guide_logging)
  amw_deploy(${app})
endforeach()
